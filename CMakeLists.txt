cmake_minimum_required(VERSION 3.22)
project(gtabr LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT ANDROID)
  set(CMAKE_BUILD_TYPE Release)
endif()

include(cmake/shaders.cmake)
gtabr_embed_shaders(GTABR_SHADER_CPP "${CMAKE_CURRENT_SOURCE_DIR}/shaders")

file(GLOB_RECURSE GTABR_CORE_SRC CONFIGURE_DEPENDS
  src/core/*.cpp src/gfx/*.cpp src/game/*.cpp)

add_library(gtabr_engine STATIC ${GTABR_CORE_SRC} ${GTABR_SHADER_CPP})
target_include_directories(gtabr_engine PUBLIC src)
target_compile_options(gtabr_engine PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
find_package(Threads REQUIRED)
target_link_libraries(gtabr_engine PUBLIC Threads::Threads)

if(ANDROID)
  # Native activity shared library (loaded by android.app.NativeActivity).
  set(NDK_GLUE "${ANDROID_NDK}/sources/android/native_app_glue")
  add_library(native_app_glue STATIC "${NDK_GLUE}/android_native_app_glue.c")
  target_include_directories(native_app_glue PUBLIC "${NDK_GLUE}")
  add_library(gtabr SHARED src/platform/android_main.cpp)
  target_link_libraries(gtabr gtabr_engine native_app_glue android log vulkan z)
  target_compile_options(gtabr PRIVATE -Wall -Wextra -Wno-unused-parameter)
  # keep ANativeActivity_onCreate exported
  set_target_properties(gtabr PROPERTIES LINK_FLAGS "-u ANativeActivity_onCreate")
  target_link_libraries(gtabr_engine PUBLIC vulkan android log z)
  target_compile_definitions(gtabr_engine PUBLIC VK_USE_PLATFORM_ANDROID_KHR=1)
else()
  find_package(Vulkan REQUIRED)
  find_package(ZLIB REQUIRED)
  target_link_libraries(gtabr_engine PUBLIC Vulkan::Vulkan ZLIB::ZLIB)
  add_executable(gtabr_headless src/platform/desktop_main.cpp src/platform/scenarios.cpp)
  target_link_libraries(gtabr_headless gtabr_engine)
  add_executable(gtabr_smoke tools/smoke_render.cpp)
  target_link_libraries(gtabr_smoke gtabr_engine)
  file(GLOB SPRITEBAKE_SRC tools/spritebake/*.cpp)
  add_executable(spritebake ${SPRITEBAKE_SRC})
  target_include_directories(spritebake PRIVATE src tools/spritebake)
  target_compile_options(spritebake PRIVATE -O3)
  find_package(Threads REQUIRED)
  target_link_libraries(spritebake Threads::Threads)
  enable_testing()
  file(GLOB GTABR_TESTS CONFIGURE_DEPENDS tests/*.cpp)
  foreach(t ${GTABR_TESTS})
    get_filename_component(tn ${t} NAME_WE)
    add_executable(${tn} ${t})
    target_link_libraries(${tn} gtabr_engine)
    add_test(NAME ${tn} COMMAND ${tn} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
  endforeach()
endif()
